// ====== CRS 7-LINES SCOPE (GitHub Pages friendly) ====== const CRS_LINES = 7; // estado let audioCtx, analyser, source, micStream; let rafId = null; let startedAt = 0; let timerInt = null; // buffers p/ “rolagem” const HISTORY = 600; // mais = mais “comprimento” horizontal const series = Array.from({ length: CRS_LINES }, () => new Array(HISTORY).fill(0)); let writeIdx = 0; // config (carregar do localStorage se existir) function getCRSConfig() { const cfg = JSON.parse(localStorage.getItem("crs_config") || "{}"); return { maxMin: Number(cfg.maxMin || 30), hz: Number(cfg.hz || 20), // 20 amostras/s é ótimo p/ visual thr: Number(cfg.thr || 0.03), // threshold base de “silêncio” }; } function fmtTime(sec) { sec = Math.max(0, Math.floor(sec)); const m = String(Math.floor(sec / 60)).padStart(2, "0"); const s = String(sec % 60).padStart(2, "0"); return `${m}:${s}`; } function setText(id, v){ const el = document.getElementById(id); if(el) el.textContent = v; } function clamp01(x){ return Math.max(0, Math.min(1, x)); } // métricas rápidas function rmsFromTimeDomain(buf) { // buf: Uint8Array [0..255], 128 = zero let sum = 0; for (let i = 0; i < buf.length; i++) { const v = (buf[i] - 128) / 128; sum += v * v; } return Math.sqrt(sum / buf.length); } function bandEnergy(freq, fromHz, toHz, sampleRate, fftSize) { const nyq = sampleRate / 2; const from = Math.floor((fromHz / nyq) * freq.length); const to = Math.floor((toHz / nyq) * freq.length); let sum = 0, n = 0; for (let i = Math.max(0, from); i <= Math.min(freq.length - 1, to); i++) { sum += freq[i]; n++; } return n ? (sum / n) / 255 : 0; } // “pitch proxy”: razão entre pico de banda média e RMS (não é pitch real, mas dá variação) function pitchProxy(freq, sampleRate, fftSize) { // pega maior pico entre 90Hz e 350Hz const nyq = sampleRate / 2; const from = Math.floor((90 / nyq) * freq.length); const to = Math.floor((350 / nyq) * freq.length); let max = 0; for (let i = from; i <= to; i++) max = Math.max(max, freq[i]); return (max / 255); } function drawScope() { const canvas = document.getElementById("scope"); if (!canvas) return; const ctx = canvas.getContext("2d"); // background suave (quase branco) ctx.clearRect(0, 0, canvas.width, canvas.height); ctx.fillStyle = "rgba(245, 255, 255, 1)"; ctx.fillRect(0, 0, canvas.width, canvas.height); // grade leve ctx.strokeStyle = "rgba(0,0,0,0.05)"; ctx.lineWidth = 1; for (let i = 1; i < 7; i++) { const y = (canvas.height / 7) * i; ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(canvas.width, y); ctx.stroke(); } // 7 linhas: offsets verticais const rowH = canvas.height / CRS_LINES; // desenha cada série for (let l = 0; l < CRS_LINES; l++) { const yMid = rowH * l + rowH * 0.5; const amp = rowH * 0.35; // cor “rotativa” sem definir paleta fixa (deixa o navegador padrão) // (se você quiser, eu defino uma paleta ciano/verde/azul bem suave depois) ctx.strokeStyle = `rgba(0,0,0,0.75)`; ctx.lineWidth = 2; ctx.beginPath(); for (let x = 0; x < HISTORY; x++) { const idx = (writeIdx + x) % HISTORY; const v = series[l][idx]; // 0..1 const y = yMid - (v - 0.5) * 2 * amp; // centraliza const px = (x / (HISTORY - 1)) * canvas.width; if (x === 0) ctx.moveTo(px, y); else ctx.lineTo(px, y); } ctx.stroke(); // legenda simples ctx.fillStyle = "rgba(0,0,0,0.45)"; ctx.font = "14px system-ui, -apple-system, Segoe UI, Roboto"; const label = [ "1) RMS (energia)", "2) Pausa curta", "3) Pausa média", "4) Pausa longa", "5) Pitch proxy", "6) Graves", "7) Médias-altas", ][l]; ctx.fillText(label, 12, rowH * l + 20); } } async function startAudio() { const cfg = getCRSConfig(); setText("maxMin", String(cfg.maxMin)); setText("audioStatus", "iniciando..."); micStream = await navigator.mediaDevices.getUserMedia({ audio: true }); audioCtx = new (window.AudioContext || window.webkitAudioContext)(); analyser = audioCtx.createAnalyser(); analyser.fftSize = 2048; analyser.smoothingTimeConstant = 0.6; source = audioCtx.createMediaStreamSource(micStream); source.connect(analyser); // timers startedAt = Date.now(); if (timerInt) clearInterval(timerInt); timerInt = setInterval(() => { const sec = (Date.now() - startedAt) / 1000; setText("timer", fmtTime(sec)); if (sec >= cfg.maxMin * 60) stopAudio(true); }, 250); setText("audioStatus", "captando"); // loop de amostragem const timeData = new Uint8Array(analyser.fftSize); const freqData = new Uint8Array(analyser.frequencyBinCount); const intervalMs = Math.max(40, Math.round(1000 / cfg.hz)); let lastSample = 0; const loop = () => { rafId = requestAnimationFrame(loop); const now = performance.now(); if (now - lastSample < intervalMs) { drawScope(); return; } lastSample = now; analyser.getByteTimeDomainData(timeData); analyser.getByteFrequencyData(freqData); const rms = clamp01(rmsFromTimeDomain(timeData) * 3.0); // ganho p/ ficar visível const thr = cfg.thr; // pause scores (demo): quanto mais perto do silêncio, maior o score const silence = clamp01((thr - rms) / thr); // 0 (falando) -> 1 (silêncio) // heurística: “curta/média/longa” por suavização diferente // (isso aqui é DEMO, mas já dá leitura visual bonita) const shortPause = silence; const midPause = clamp01( (silence * 0.7) + (series[2][(writeIdx-1+HISTORY)%HISTORY] * 0.3) ); const longPause = clamp01( (silence * 0.5) + (series[3][(writeIdx-1+HISTORY)%HISTORY] * 0.5) ); const pp = clamp01(pitchProxy(freqData, audioCtx.sampleRate, analyser.fftSize)); const low = clamp01(bandEnergy(freqData, 20, 120, audioCtx.sampleRate, analyser.fftSize)); const midHigh = clamp01(bandEnergy(freqData, 800, 3500, audioCtx.sampleRate, analyser.fftSize)); // escreve 7 linhas (0..1) series[0][writeIdx] = rms; // 1 RMS series[1][writeIdx] = shortPause; // 2 pausa curta series[2][writeIdx] = midPause; // 3 pausa média series[3][writeIdx] = longPause; // 4 pausa longa series[4][writeIdx] = pp; // 5 pitch proxy series[5][writeIdx] = low; // 6 graves series[6][writeIdx] = midHigh; // 7 medias-altas writeIdx = (writeIdx + 1) % HISTORY; // aqui você pode ir gravando também no localStorage p/ relatório // (depois eu te mando o relatório bonito com “sumário” e “amostras”) drawScope(); }; loop(); } function stopAudio(auto = false) { if (rafId) cancelAnimationFrame(rafId); rafId = null; if (timerInt) clearInterval(timerInt); timerInt = null; try { if (source) source.disconnect(); } catch(e){} try { if (analyser) analyser.disconnect?.(); } catch(e){} try { if (audioCtx) audioCtx.close(); } catch(e){} if (micStream) { micStream.getTracks().forEach(t => t.stop()); } setText("audioStatus", auto ? "encerrado (tempo)" : "encerrado"); } // binds document.addEventListener("DOMContentLoaded", () => { const a = document.getElementById("btnStartAudio"); const b = document.getElementById("btnStopAudio"); if (a) a.addEventListener("click", startAudio); if (b) b.addEventListener("click", () => stopAudio(false)); });
